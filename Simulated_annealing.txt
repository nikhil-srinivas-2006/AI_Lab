import random
import math

def cost(state):
    c = 0
    for i in range(8):
        for j in range(i + 1, 8):
            if abs(state[i] - state[j]) == abs(i - j):
                c += 1
    return c

def simulated_annealing():
    state = random.sample(range(8), 8)
    T = 100
    cooling = 0.99

    while T > 0.01:
        current_cost = cost(state)

        if current_cost == 0:
            return state

        i, j = random.sample(range(8), 2)
        neighbor = state.copy()
        neighbor[i], neighbor[j] = neighbor[j], neighbor[i]

        delta = cost(neighbor) - current_cost

        if delta < 0 or random.random() < math.exp(-delta / T):
            state = neighbor

        T *= cooling

    return state if cost(state) == 0 else None

solution = simulated_annealing()

if solution:
    print("Solution:")
    for i in range(8):
        print(" ".join("Q" if solution[j] == i else "." for j in range(8)))
    print("Cost:", cost(solution))
else:
    print("Solution not found. Run again.")
